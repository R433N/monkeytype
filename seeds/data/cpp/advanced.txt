template <typename T>
T maxOf(const std::vector<T>& items) {
    T best = items[0];
    for (const T& item : items) {
        if (item > best) {
            best = item;
        }
    }
    return best;
}

class Rectangle {
public:
    Rectangle(double w, double h) : width(w), height(h) {}

    double area() const {
        return width * height;
    }

private:
    double width;
    double height;
};
